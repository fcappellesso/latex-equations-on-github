\begin{equation}\label{eq:trace-operator}
T \colon W^{1,p}(\Omega) \to L^p(\partial\Omega),
\end{equation}
\begin{equation}\label{eq:trace-estimate}
\|Tu\|_{L^p(\partial\Omega)} \leq C\,\|u\|_{W^{1,p}(\Omega)},
\qquad \forall\, u \in W^{1,p}(\Omega).
\end{equation}
